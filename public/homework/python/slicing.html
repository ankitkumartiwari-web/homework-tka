a ="Ankit Tiwari"

#slicing operations
print(a[0:5])  # Output: Ankit
print(a[6:12]) # Output: Tiwari
print(a[0:12:2]) # Output: Aitri
print(a[-6:-1]) # Output: Tiwari
print(a[::-1]) # Output: irawiT tiknA
print(a[::3]) # Output: Aii
print(a[1:10:2]) # Output: nki i
print(a[5:]) # Output: Tiwari
print(a[:5]) # Output: Ankit
print(a[::]) # Output: Ankit Tiwari
